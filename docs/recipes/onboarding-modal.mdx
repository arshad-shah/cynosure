# Onboarding modal

A modal dialog that walks a new user through three short steps. Uses
`Dialog`, `Progress`, and controlled state to prevent accidental dismiss
mid-flow.

```tsx
import { useState } from 'react';
import {
  Button,
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
  Progress,
  Stack,
  Text,
} from '@lumen/react';

const STEPS = [
  {
    title: 'Welcome',
    description: 'Quick tour of Lumen — under 30 seconds.',
  },
  {
    title: 'Your workspace',
    description: 'We built you a starter project. Have a look around.',
  },
  {
    title: 'Invite your team',
    description: 'Or skip and do it later.',
  },
] as const;

export function OnboardingModal({
  open,
  onComplete,
}: {
  open: boolean;
  onComplete: () => void;
}) {
  const [step, setStep] = useState(0);
  const current = STEPS[step];
  const progress = ((step + 1) / STEPS.length) * 100;

  return (
    <Dialog open={open}>
      <DialogContent>
        <Stack gap="4">
          <Progress
            value={progress}
            aria-label={`Step ${step + 1} of ${STEPS.length}`}
          />

          <DialogHeader>
            <DialogTitle>{current.title}</DialogTitle>
            <DialogDescription>{current.description}</DialogDescription>
          </DialogHeader>

          <Text>
            {/* Step body goes here — swap in an <img>, an illustration,
                or a short video as appropriate. */}
            Step {step + 1} content.
          </Text>

          <DialogFooter>
            {step > 0 && (
              <Button
                variant="outline"
                onClick={() => setStep((s) => s - 1)}
              >
                Back
              </Button>
            )}
            <Button
              onClick={() => {
                if (step < STEPS.length - 1) {
                  setStep((s) => s + 1);
                } else {
                  onComplete();
                }
              }}
            >
              {step < STEPS.length - 1 ? 'Next' : 'Finish'}
            </Button>
          </DialogFooter>
        </Stack>
      </DialogContent>
    </Dialog>
  );
}
```

## Why a controlled `Dialog`

Leaving the dialog uncontrolled lets users press `Esc` mid-onboarding,
which can leave them in a half-set-up state. Keeping `open` controlled +
providing no close button means they must finish or explicitly skip.
